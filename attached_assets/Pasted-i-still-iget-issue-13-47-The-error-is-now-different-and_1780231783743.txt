// BEFORE ❌
<SEOHead
  title="Trouvez Votre Bien au Maroc"

  ogImage={`${SITE_URL}/og-image.png`}   // ← DELETE THIS LINE
  jsonLd={jsonLd}
/>

// AFTER ✅
<SEOHead
  title="Trouvez Votre Bien au Maroc"
  description="La principale plateforme immobilière du Maroc. ..."
  canonical={SITE_URL}
  jsonLd={jsonLd}
